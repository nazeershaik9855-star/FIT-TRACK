    goals: "fittrackGoals",

    foods: "fittrackFoods",

    current: "fittrackCurrentUser"
};

const today = () => new Date().toISOString().slice(0, 10);

const load = (k, d = []) =>
    JSON.parse(localStorage.getItem(k) || JSON.stringify(d));

const save = (k, v) =>
    localStorage.setItem(k, JSON.stringify(v));

function toast(msg) {
    const t = document.getElementById("toast");

    t.textContent = msg;
    t.style.display = "block";

    setTimeout(() => {
        t.style.display = "none";
    }, 2500);
}

function showLogin(msg = "", ok = false) {
    document.getElementById("loginCard").style.display = "block";
    document.getElementById("signupCard").style.display = "none";

    if (msg) {
        const m = document.getElementById("loginMessage");

        m.textContent = msg;
        m.className = "auth-message " + (ok ? "success" : "error");
    }
}

function showSignup() {
    document.getElementById("loginCard").style.display = "none";
    document.getElementById("signupCard").style.display = "block";
}

function signupUser(e) {
    e.preventDefault();

    let name = signupName.value.trim();
    let username = signupEmail.value.trim();
    let password = signupPassword.value;
    let confirm = signupConfirm.value;

    let users = load(KEYS.users, []);

    if (password !== confirm)
        return setMsg(
            "signupMessage",
            "Passwords do not match.",
            "error"
        );

    if (
        users.some(
            u => u.username.toLowerCase() === username.toLowerCase()
        )
    )
        return setMsg(
            "signupMessage",
            "Account already exists.",
            "error"
        );

    users.push({
        id: Date.now().toString(),
        name,
        username,
        password,
        role: "user",
        createdAt: today()
    });

    save(KEYS.users, users);

    e.target.reset();

    showLogin(
        "Account created successfully. Please login.",
        true
    );
}

function setMsg(id, msg, cls) {
    const x = document.getElementById(id);

    x.textContent = msg;
    x.className = "auth-message " + cls;
}

function loginUser(e) {
    e.preventDefault();

    let username = loginEmail.value.trim().toLowerCase();
    let password = loginPassword.value;

    let user = load(KEYS.users, []).find(
        u =>
            u.username.toLowerCase() === username &&
            u.password === password
    );

    if (!user)
        return setMsg(
            "loginMessage",
            "Invalid username/email or password.",
            "error"
        );

    localStorage.setItem(
        KEYS.current,
        JSON.stringify(user)
    );

    openApp(user);

    setMsg("loginMessage", "", "");
}

function openApp(user) {
    document.getElementById("loginScreen").style.display = "none";
    document.getElementById("app").style.display = "flex";

    loggedUser.textContent = user.name;

    roleLabel.innerHTML =
        user.role === "admin"
            ? '<span class="admin-badge">ADMIN</span>'
            : '<span class="user-badge">USER</span>';

    if (user.role === "admin")
        renderAdmin();
    else
        renderUser();
}

function logoutUser() {
    localStorage.removeItem(KEYS.current);

    document.getElementById("app").style.display = "none";
    document.getElementById("loginScreen").style.display = "flex";

    showLogin();
}

function currentUser() {
    return JSON.parse(
        localStorage.getItem(KEYS.current) || "null"
    );
}

function nav(items) {
    document.getElementById("nav").innerHTML =
        items
            .map(
                (x, i) =>
                    `<a href="#"
                        class="${i === 0 ? "active" : ""}"
                        onclick="showPage('${x.id}',this);return false">
                        ${x.icon} ${x.name}
                    </a>`
            )
            .join("");
}

function goalRow(name, val, goal) {
    let p = Math.min(
        100,
        Math.round((Number(val) || 0) / goal * 100)
    );

    return `
        <div class="goal">

            <div class="goal-head">
                <span>${name}</span>
                <b>${val}/${goal}</b>
            </div>

            <div class="progress">
                <div class="bar" style="width:${p}%"></div>
            </div>

        </div>
    `;
}

function chartHTML() {
    let h = load(KEYS.history, [])
        .filter(x => x.userId === currentUser().id)
        .slice(-7);

    if (!h.length)
        return "<p class='small'>No activity history yet.</p>";

    let max = Math.max(
        ...h.map(x => x.steps),
        1000
    );

    return `
        <div class="chart">

            ${
                h
                    .map(
                        x => `
                            <div class="barcol">

                                <b>${x.steps}</b>

                                <div class="chartbar"
                                    style="height:${Math.max(
                                        5,
                                        x.steps / max * 180
                                    )}px">
                                </div>

                                <span>${x.date.slice(5)}</span>

                            </div>
                        `
                    )
                    .join("")
            }

        </div>
    `;
}

function addWorkout(e) {
    e.preventDefault();

    let a = load(KEYS.workouts, []);

    a.push({
        id: Date.now().toString(),
        userId: currentUser().id,
        type: wType.value,
        duration: Number(wDuration.value),
        calories: Number(wCalories.value),
        date: today()
    });

    save(KEYS.workouts, a);

    let d = getTodayData();

    d.workout += Number(wDuration.value);
    d.calories += Number(wCalories.value);

    saveToday(d);
    saveHistory(d);

    toast("Workout saved");

    userActivity();
}

function saveDaily() {
    let d = getTodayData();

    d.steps = Number(aSteps.value || d.steps);
    d.water = Number(aWater.value || d.water);
    d.sleep = Number(aSleep.value || d.sleep);
    d.distance = Number(aDistance.value || d.distance);

    saveToday(d);
    saveHistory(d);

    toast("Daily data saved");

    userDashboard();
}

function getTodayData() {
    let all = load(KEYS.data, {});

    return (
        all[today()] || {
            steps: 0,
            calories: 0,
            distance: 0,
            heartRate: null,
            water: 0,
            workout: 0,
            sleep: 0
        }
    );
}

function saveToday(d) {
    let all = load(KEYS.data, {});

    all[today()] = d;

    save(KEYS.data, all);
}

function saveHistory(d) {
    let h = load(KEYS.history, [])
        .filter(
            x =>
                !(
                    x.userId === currentUser().id &&
                    x.date === today()
                )
        );

    h.push({
        userId: currentUser().id,
        date: today(),
        steps: d.steps,
        calories: d.calories,
        workout: d.workout
    });

    save(KEYS.history, h);
}

async function connectWearable() {
    if (!navigator.bluetooth) {
        toast(
            "Web Bluetooth is not supported in this browser."
        );
        return;
    }

    try {
        const device =
            await navigator.bluetooth.requestDevice({
                acceptAllDevices: true,
                optionalServices: [
                    "heart_rate",
                    "battery_service",
                    "device_information"
                ]
            });

        const server = await device.gatt.connect();

        let d = getTodayData();

        try {
            const s =
                await server.getPrimaryService("heart_rate");

            const c =
                await s.getCharacteristic(
                    "heart_rate_measurement"
                );

            c.addEventListener(
                "characteristicvaluechanged",
                e => {
                    let v = e.target.value;

                    let flags = v.getUint8(0);

                    let hr =
                        flags & 1
                            ? v.getUint16(1, true)
                            : v.getUint8(1);

                    d = getTodayData();

                    d.heartRate = hr;

                    saveToday(d);

                    userDashboard();
                }
            );

            await c.startNotifications();

        } catch (e) {}

        try {
            const s =
                await server.getPrimaryService(
                    "battery_service"
                );

            const c =
                await s.getCharacteristic(
                    "battery_level"
                );

            let v = await c.readValue();

            toast(
                "Connected: " +
                device.name +
                " | Battery " +
                v.getUint8(0) +
                "%"
            );

        } catch (e) {

            toast(
                "Connected: " +
                (device.name || "BLE device")
            );
        }

        const dot =
            document.getElementById("statusDot");

        if (dot)
            dot.classList.add("on");

        const ct =
            document.getElementById("connectionText");

        if (ct)
            ct.textContent =
                "Connected: " +
                (device.name || "BLE device");

    } catch (e) {

        toast(
            "Bluetooth connection cancelled or failed."
        );
    }
}

function adminUsers() {

                    </tbody>

                </table>

            </div>

        </div>

}

function userRows(users) {
    return users
        .map(
            u => `
                <tr>

                    <td>${escapeHtml(u.name)}</td>

                    <td>${escapeHtml(u.username)}</td>

                    <td>
                        ${
                            u.role === "admin"
                                ? '<span class="admin-badge">ADMIN</span>'
                                : '<span class="user-badge">USER</span>'
                        }
                    </td>

                    <td>${u.createdAt || "-"}</td>

                    <td>

                        ${
                            u.role === "user"
                                ? `
                                    <button
                                        class="danger"
                                        onclick="removeUser('${u.id}')">
                                        Remove
                                    </button>
                                `
                                : "Protected"
                        }

                    </td>

                </tr>
            `
        )
        .join("");
}

function filterUsers() {
    let q =
        document
            .getElementById("userSearch")
            .value
            .toLowerCase();

    let a = load(KEYS.users, [])
        .filter(u =>
            (u.name + " " + u.username + " " + u.role)
                .toLowerCase()
                .includes(q)
        );

    document.getElementById("userRows").innerHTML =
        userRows(a);
}

function removeUser(id) {
    if (!confirm("Remove this user?"))
        return;

    save(
        KEYS.users,
        load(KEYS.users, [])
            .filter(u => u.id !== id)
    );

    save(
        KEYS.workouts,
        load(KEYS.workouts, [])
            .filter(x => x.userId !== id)
    );

    save(
        KEYS.goals,
        load(KEYS.goals, [])
            .filter(x => x.userId !== id)
    );

    toast("User removed");

    adminUsers();
}

function adminReports() {
    let users =
        load(KEYS.users, [])
            .filter(u => u.role === "user");

    let w = load(KEYS.workouts, []);
    let h = load(KEYS.history, []);

    let rows = users
        .map(u => {

            let hw =
                h.filter(x => x.userId === u.id);

            let ww =
                w.filter(x => x.userId === u.id);

            let steps =
                hw.reduce(
                    (s, x) => s + Number(x.steps || 0),
                    0
                );

            return `
                <tr>

                    <td>
                        ${escapeHtml(u.name)}
                    </td>

                    <td>
                        ${ww.length}
                    </td>

                    <td>
                        ${steps.toLocaleString()}
                    </td>

                    <td>
                        ${ww.reduce(
                            (s, x) =>
                                s + Number(x.calories || 0),
                            0
                        )}
                    </td>

                </tr>
            `;
        })
        .join("");

    main.innerHTML = `

        <div class="top">

            <div>

                <h1>Reports & Activity</h1>

                <div class="date">
                    Overall user activity and goal progress
                </div>

            </div>

        </div>

        <div class="grid">

            <div class="card">

                <div class="label">
                    Registered Users
                </div>

                <div class="metric">
                    ${users.length}
                </div>

            </div>

            <div class="card">

                <div class="label">
                    Total Workouts
                </div>

                <div class="metric">
                    ${w.filter(x => !x.library).length}
                </div>

            </div>

            <div class="card">

                <div class="label">
                    Total Steps Recorded
                </div>

                <div class="metric">
                    ${h.reduce(
                        (s, x) =>
                            s + Number(x.steps || 0),
                        0
                    ).toLocaleString()}
                </div>

            </div>

            <div class="card">

                <div class="label">
                    Workout Calories
                </div>

                <div class="metric">
                    ${w
                        .filter(x => !x.library)
                        .reduce(
                            (s, x) =>
                                s + Number(x.calories || 0),
                            0
                        )}
                </div>

            </div>

        </div>

        <div class="card section">

            <h2 class="section-title">
                User Activity Report
            </h2>

            <div class="table-wrap">

                <table class="table">

                    <tr>
                        <th>User</th>
                        <th>Workouts</th>
                        <th>Steps</th>
                        <th>Calories Burned</th>
                    </tr>

                    ${
                        rows ||
                        `
                            <tr>
                                <td colspan="4">
                                    No user activity yet.
                                </td>
                            </tr>
                        `
                    }

                </table>

            </div>

        </div>
    `;
}

function escapeHtml(s) {
    return String(s ?? "")
        .replace(
            /[&<>"']/g,
            m =>
                ({
                    "&": "&amp;",
                    "<": "&lt;",
                    ">": "&gt;",
                    '"': "&quot;",
                    "'": "&#039;"
                }[m])
        );
}
